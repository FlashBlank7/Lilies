def integrate_gold_price(sales_data, gold_api_data):
    """将销售数据与黄金价格关联"""
    # 基于商品材质(ZMETAL)和销售日期(FKDAT)关联
    gold_sales = sales_data[sales_data['ZMETAL'].str.contains('金', na=False)]
    merged_data = pd.merge(gold_sales, gold_api_data,
